Client from a CDN

<script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2.117.2"></script>
<script>
  const db = window.supabase.createClient(URL, PUBLISHABLE_KEY);
</script>

Pin an exact version. Do not call the client supabase: that name is the global from the script.

Auth

await db.auth.signInWithPassword({ email, password });
await db.auth.signOut();
const { data: { session } } = await db.auth.getSession();
db.auth.onAuthStateChange((event, session) => { /* keep it synchronous */ });

Read and write

// read, with a related table embedded through its foreign key
await db.from('demo_tasks')
  .select('id, title, demo_projects(name)')
  .order('created_at', { ascending: false });

await db.from('demo_tasks').select('*').eq('id', taskId);
await db.from('demo_tasks').insert({ project_id, owner_id, title });
await db.from('demo_tasks').delete().eq('id', taskId);   // needs a grant + policy

Every call returns { data, error }; check error.code.

Realtime

alter publication supabase_realtime add table public.demo_tasks;
const channel = db.channel('tasks-live')
  .on('postgres_changes',
      { event: 'INSERT', schema: 'public', table: 'demo_tasks' },
      payload => refresh())
  .subscribe(state => console.log(state));   // 'SUBSCRIBED'
db.removeChannel(channel);

event: INSERT, UPDATE, DELETE or *. DELETE events are not filtered by RLS [8].

Tables and keys

create table public.demo_tasks (
  id         uuid primary key default gen_random_uuid(),
  project_id uuid not null references public.demo_projects(id),
  owner_id   uuid not null references auth.users(id) on delete cascade,
  title      text not null check (char_length(btrim(title)) between 1 and 120),
  created_at timestamptz not null default now()
);

Grants: which operations

revoke all on public.demo_tasks from public, anon, authenticated;
grant select, insert on public.demo_tasks to authenticated;

RLS policies: which rows

alter table public.demo_tasks enable row level security;

create policy "read own" on public.demo_tasks
  for select to authenticated
  using ((select auth.uid()) = owner_id);

create policy "insert own" on public.demo_tasks
  for insert to authenticated
  with check ((select auth.uid()) = owner_id);

create policy "delete own" on public.demo_tasks
  for delete to authenticated
  using ((select auth.uid()) = owner_id);

using filters rows silently; a failed with check raises 42501. Wrap auth.uid() in select for speed [4].

  • Free, $0: 500 MB database, 50,000 monthly active users, 5 GB egress; pauses after a week idle.

  • Export any time: supabase db dump or pg_dump.

Safety checklist

  • RLS enabled on every table the API can reach
  • A policy for each operation you grant
  • Only the publishable key in the browser
  • config.js with real values not committed
  • Sign-ups closed for test projects
  • Test with a second user in a private window
